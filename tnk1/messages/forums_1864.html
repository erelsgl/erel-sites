פרשת: כי תבוא-כיצד נתברך בארצנו-בברכות ועליונות על האויבים?/ אהובה קליין

מאת: אהובה קליין

פרשת כי תבוא – כיצד נתברך בארצנו בברכות ועליונות על האויבים?
מאת: אהובה קליין.

"וְהָיָה, כִּי תָבוֹא אֶל הָאָרֶץ, אֲשֶׁר יְהוָה אֱלֹהֶיךָ, נֹתֵן לְךָ נַחֲלָה; וִירִשְׁתָּהּ, וְיָשַׁבְתָּ בָּהּ. וְלָקַחְתָּ מֵרֵאשִׁית כָּל פְּרִי הָאֲדָמָה, אֲשֶׁר תָּבִיא מֵאַרְצְךָ אֲשֶׁר יְהוָה אֱלֹהֶיךָ נֹתֵן לָךְ--וְשַׂמְתָּ בַטֶּנֶא; וְהָלַכְתָּ, אֶל־ הַמָּקוֹם, אֲשֶׁר יִבְחַר יְהוָה אֱלֹהֶיךָ, לְשַׁכֵּן שְׁמוֹ שָׁם. וּבָאתָ, אֶל הַכֹּהֵן, אֲשֶׁר יִהְיֶה, בַּיָּמִים הָהֵם; וְאָמַרְתָּ אֵלָיו, הִגַּדְתִּי הַיּוֹם לַיהוָה אֱלֹהֶיךָ, כִּי בָאתִי אֶל־ הָאָרֶץ, אֲשֶׁר נִשְׁבַּע יְהוָה לַאֲבֹתֵינוּ לָתֶת לָנוּ. וְלָקַח הַכֹּהֵן הַטֶּנֶא, מִיָּדֶךָ; וְהִנִּיחוֹ--לִפְנֵי, מִזְבַּח יְהוָה אֱלֹהֶיךָ. וְעָנִיתָ וְאָמַרְתָּ לִפְנֵי יְהוָה אֱלֹהֶיךָ, אֲרַמִּי אֹבֵד אָבִי, וַיֵּרֶד מִצְרַיְמָה, וַיָּגָר שָׁם בִּמְתֵי מְעָט; וַיְהִי ־שָׁם, לְגוֹי גָּדוֹל עָצוּם וָרָב...." [דברים כ"ו, א']
בהפטרה קוראים: את נבואת ישעיהו הנביא, להלן: קטע קצר מההפטרה:
"קוּמִי אוֹרִי, כִּי בָא אוֹרֵךְ; וּכְבוֹד יְהוָה, עָלַיִךְ זָרָח. כִּי־הִנֵּה הַחֹשֶׁךְ יְכַסֶּה אֶרֶץ, וַעֲרָפֶל לְאֻמִּים; וְעָלַיִךְ יִזְרַח יְהוָה, וּכְבוֹדוֹ עָלַיִךְ יֵרָאֶה. וְהָלְכוּ גוֹיִם, לְאוֹרֵךְ; וּמְלָכִים, לְנֹגַהּ זַרְחֵךְ. שְׂאִי־ סָבִיב עֵינַיִךְ, וּרְאִי--כֻּלָּם, נִקְבְּצוּ בָאוּ לָךְ; בָּנַיִךְ מֵרָחוֹק יָבֹאוּ, וּבְנוֹתַיִךְ עַל־צַד תֵּאָמַנָה. אָז תִּרְאִי וְנָהַרְתְּ, וּפָחַד וְרָחַב לְבָבֵךְ: כִּי יֵהָפֵךְ עָלַיִךְ הֲמוֹן יָם, חֵיל גּוֹיִם יָבֹאוּ לָךְ".[ישעיהו פרק: ס, א'- ו']
השאלות הן:
א] כיצד נגיע למעלת השמחה גם בימי מלחמה?
ב] איך נבטיח ניצחון של עם קטן מול אויבים רבים?
ג] מה מנבא ישעיהו הנביא על עתיד עם ישראל?
תשובות.
היכולת לממש את השמחה גם בימי מלחמה.
"אור החיים" סובר כי המילה: "וְהָיָה"- היא מלשון: שמחה - מכאן ניתן ללמוד שאין לשמוח, אלא בישיבת הארץ כדברי דוד המלך:
"...בְּשׁ֣וּב יְ֭הוָה אֶת שִׁיבַ֣ת צִיּ֑וֹן הָ֝יִ֗ינוּ כְּחֹֽלְמִֽים׃ אָ֤ז יִמָּלֵ֪א שְׂח֡וֹק פִּינוּ֮ וּלְשׁוֹנֵ֪נוּ רִ֫נָּ֥ה אָ֭ז יֹֽאמְר֣וּ בַגּוֹיִ֑ם הִגְדִּ֥יל יְ֝הוָ֗ה לַֽעֲשׂ֥וֹת עִם־ אֵֽלֶּה׃ הִגְדִּ֣יל יְ֭הוָה לַֽעֲשׂ֥וֹת עִמָּ֗נוּ הָיִ֥ינוּ שְׂמֵחִֽים"׃[ תהלים קכ"ו, א'- ד'] על מנת להגיע למעלת השמחה- צריך יהודי לפתוח את שערי המחשבה – והמילה :"מחשבה" בחילופי אותיות היא: בשמחה.
אל השמחה נגיע כאשר לא נחשוב – כי הכול מובן מאליו ,עצם הישיבה בארץ ישראל גורמת ליהודי שמחה רבה- זו מצווה מיוחדת לפי - שכל גופו טובל בארץ ישראל, כאשר היהודי היה מביא ביכורים בתקופת המקדש עצם העניין שהיה לוקח "מֵרֵאשִׁית כָּל־ פְּרִי הָאֲדָמָה" הרי כל "רֵאשִׁית"- כל דבר חדש מעורר אצל האדם שמחה , במיוחד כאשר לוקח את הביכורים אל בית המקדש - עצם ההגעה לבית המקדש היא זכות גדולה - ושמחה גדולה:
כדברי דוד המלך: "שָׂ֭מַחְתִּי בְּאֹֽמְרִ֣ים לִ֑י בֵּ֖ית יְהוָ֣ה נֵלֵֽךְ" [תהלים קכ"ב]
כיצד היו עולים לירושלים?
חז"ל מסבירים: "כל העיירות שבמעמד מתכנסות לעיר של מעמד, ולנים ברחובה של עיר- ולא היו נכנסין לבתים. ואילו למשכים [אלה שרצו לבקר השכם היינו - מוקדם] היה הממונה אומר: "קוּמוּ וְנַעֲלֶה צִיּוֹן, אֶל יְהוָה אֱלֹהֵינוּ" [ירמיהו ל"א, ה'].
הקרובים מביאים תאנים וענבים והרחוקים מביאים גרוגרות וצימוקים - והשור הולך לפניהם - קרניו מצופות זהב ועטרה של זית בראשו - החליל מכה לפניהם - עד שמגיעים קרוב לירושלים, שלחו לפניהם ועטרו את בכוריהם. הפחות - הסגנים והגזברים- יוצאים לקראתם לפי כבוד הנכנסים היו יוצאים. וכל בעלי אומנויות שבירושלים- עומדים ושואלים בשלומם: אחינו אנשי מקום פלוני, באתם לשלום"?
לעניות דעתי, דווקא בימים אלה כאשר אנחנו קרובים לשער ראש השנה כולנו מייחלים לשנה חדשה טובה ושלווה. חייבים אנו ,ביתר שאת - להתחבר לספר התנ"ך ובמיוחד ללמוד את פרשיות השבוע – מתוך חמישה חומשי התורה -בכל בית בישראל - כי התורה היא: מקור השמחה.
גם מלך שהיה מולך על עם ישראל היה חייב להיות מחובר לשני ספרי תורה: האחד- כאשר הוא בארמונו והשני: כאשר הוא – יוצא לרשות הרבים כנאמר: "וְהָיָה כְשִׁבְתּוֹ, עַל כִּסֵּא מַמְלַכְתּוֹ--וְכָתַב לוֹ אֶת מִשְׁנֵה הַתּוֹרָה הַזֹּאת, עַל סֵפֶר, מִלִּפְנֵי, הַכֹּהֲנִים הַלְוִיִּם. וְהָיְתָה עִמּוֹ, וְקָרָא בוֹ כָּל יְמֵי חַיָּיו--לְמַעַן יִלְמַד, לְיִרְאָה אֶת יְהוָה אֱלֹהָיו, לִשְׁמֹר אֶת־ כָּל־ דִּבְרֵי הַתּוֹרָה הַזֹּאת וְאֶת ־ הַחֻקִּים הָאֵלֶּה, לַעֲשֹׂתָם. לְבִלְתִּי רוּם לְבָבוֹ מֵאֶחָיו, וּלְבִלְתִּי סוּר מִן־ הַמִּצְוָה יָמִין וּשְׂמֹאול-לְמַעַן יַאֲרִיךְ יָמִים עַל־ מַמְלַכְתּוֹ הוּא וּבָנָיו, בְּקֶרֶב יִשְׂרָאֵל". [דברים י"ז, י"ח-כ']
מכאן, שהתורה היא: מקור השמחה
"וְהָיָה כְשִׁבְתּוֹ, עַל כִּסֵּא מַמְלַכְתּוֹ--וְכָתַב לוֹ אֶת־ מִשְׁנֵה הַתּוֹרָה"
זאת בנוסף על הישיבה בארץ ישראל. הרי אין השכינה שורה , אלא מתוך שמחה.
בארצנו הקדושה אלוקים מביט בבניו יומם וליל - כדברי דוד המלך באומרו:
"הִנֵּ֣ה לֹֽא יָ֭נוּם וְלֹ֣א יִישָׁ֑ן שׁ֝וֹמֵ֗ר יִשְׂרָאֵֽל׃ה יְהוָ֥ה שֹֽׁמְרֶ֑ךָ יְהוָ֥ה צִ֝לְּךָ֗ עַל־ יַ֥ד יְמִינֶֽךָ"׃[תהלים קכ"א, ד'- ו']
ונאמר: "אֶרֶץ, אֲשֶׁר יְהוָה אֱלֹהֶיךָ דֹּרֵשׁ אֹתָהּ: תָּמִיד, עֵינֵי יְהוָה אֱלֹהֶיךָ בָּהּ--מֵרֵשִׁית הַשָּׁנָה, וְעַד אַחֲרִית שָׁנָה".[דברים י"א, י"ב]
חייב כל יהודי לדעת: כי ארץ ישראל מרוממת מכל הארצות - ארץ שהובטחה עוד לאברהם אבי האומה כפי שנאמר:
"בַּיּוֹם הַהוּא, כָּרַת יְהוָה אֶת אַבְרָם--בְּרִית לֵאמֹר: לְזַרְעֲךָ, נָתַתִּי אֶת הָאָרֶץ הַזֹּאת, מִנְּהַר מִצְרַיִם, עַד הַנָּהָר הַגָּדֹל נְהַר פְּרָת" [בראשית ט"ו. י"ח]
בימים אלה חייבים אנו להתאחד כאיש אחד בלב אחד.
כי אחדות ואהבת חינם - היא כוח עוצמתי-להשגת הצלחה ובנוסף עלינו להיות מודעים- כי זו זכות גדולה לשבת בארץ ישראל כאשר אנחנו מקיימים את המצוות בין אדם לחברו ובין אדם למקום.
הבטחת ניצחון של עם קטן כנגד אויבים רבים.
ישנם אנשים הסוברים - כי התורה היא כה עתיקה שאינה מתאימה לעולמנו כיום וזאת כאשר האנושות הגיעה להישגים כה עוצמתיים- אניות פאר, גורדי שחקים , בינה מלאכותית.
בתקופה זו: להגיע לארצות רחוקות מעבר לים הוא בהישג יד ממש. כאן ישנה סכנה של גאווה ועל כך התורה מזהירה אותנו - כן אותה תורה עתיקת יומין - כחדשה ממש גם בתקופתנו פונה ומדגישה:
"הִשָּׁמֶר לְךָ, פֶּן תִּשְׁכַּח אֶת יְהוָה אֱלֹהֶיךָ, לְבִלְתִּי שְׁמֹר מִצְוÊתָיו וּמִשְׁפָּטָיו וְחֻקֹּתָיו, אֲשֶׁר אָנֹכִי מְצַוְּךָ הַיּוֹם. פֶּן תֹּאכַל, וְשָׂבָעְתָּ; וּבָתִּים טֹבִים תִּבְנֶה, וְיָשָׁבְתָּ. וּבְקָרְךָ וְצֹאנְךָ יִרְבְּיֻן, וְכֶסֶף וְזָהָב יִרְבֶּה לָּךְ; וְכֹל אֲשֶׁר לְךָ, יִרְבֶּה. וְרָם, לְבָבֶךָ; וְשָׁכַחְתָּ אֶת יְהוָה אֱלֹהֶיךָ, הַמּוֹצִיאֲךָ מֵאֶרֶץ מִצְרַיִם מִבֵּית עֲבָדִים. הַמּוֹלִיכְךָ בַּמִּדְבָּר הַגָּדֹל וְהַנּוֹרָא, נָחָשׁ שָׂרָף וְעַקְרָב, וְצִמָּאוֹן, אֲשֶׁר אֵין מָיִם; הַמּוֹצִיא לְךָ מַיִם, מִצּוּר הַחַלָּמִישׁ. הַמַּאֲכִלְךָ מָן בַּמִּדְבָּר, אֲשֶׁר לֹא יָדְעוּן אֲבֹתֶיךָ: לְמַעַן עַנֹּתְךָ, וּלְמַעַן נַסֹּתֶךָ לְהֵיטִבְךָ, בְּאַחֲרִיתֶךָ. וְאָמַרְתָּ, בִּלְבָבֶךָ: כֹּחִי וְעֹצֶם יָדִי, עָשָׂה לִי אֶת הַחַיִל הַזֶּה".[דברים ח', י"א- י"ח]
התורה מזהירה את עם ישראל- מהשכחה "פֶּן־ תִּשְׁכַּח אֶת ־יְהוָה אֱלֹהֶיךָ",
המילה: "תִּשְׁכַּח" מורכבת מהמילים: תש - כוח!
מכאן לומדים: אימתי עמלק מגיע? כאשר תש כוחנו מהתורה חלילה!
כנאמר:" זָכוֹר, אֵת אֲשֶׁר־ עָשָׂה לְךָ עֲמָלֵק, בַּדֶּרֶךְ, בְּצֵאתְכֶם מִמִּצְרָיִם. אֲשֶׁר קָרְךָ בַּדֶּרֶךְ, וַיְזַנֵּב בְּךָ כָּל־ הַנֶּחֱשָׁלִים אַחֲרֶיךָ--וְאַתָּה, עָיֵף וְיָגֵעַ; וְלֹא יָרֵא, אֱלֹהִים.:[דברים כ"ה, י"ח]
לכן כאשר מופיע אדם החפץ להנהיג את עם ישראל, ומבטיח הרים וגבעות ומנסה להתרחק מהתורה – מיראת שמים ויהדות - חלילה לעם שלנו שיגיע לכיסא המנהיגות – במיוחד בימים אלה - כאשר כל אזרח קשוב לאמצעי התקשורת השונים. במידה והוא אינו מחובר לתורה הקדושה יכול להגיע האזרח הקטן לבלבול ולבחור בתומכי האויבים!
ארץ ישראל יכולה להינצל מכל האויבים סביב – אם תבחר בדרך התורה הקדושה.
השליחות של עם ישראל = להיות קדוש ולהאיר באמצעות התורה לכל אומות העולם. כנאמר:
"וְאַתֶּם תִּהְיוּ ־לִי מַמְלֶכֶת כֹּהֲנִים, וְגוֹי קָדוֹשׁ: אֵלֶּה, הַדְּבָרִים, אֲשֶׁר תְּדַבֵּר, אֶל־בְּנֵי יִשְׂרָאֵל. וַיָּבֹא מֹשֶׁה, וַיִּקְרָא לְזִקְנֵי הָעָם; וַיָּשֶׂם לִפְנֵיהֶם, אֵת כָּל־ הַדְּבָרִים הָאֵלֶּה, אֲשֶׁר צִוָּהוּ, יְהוָה. וַיַּעֲנוּ כָל־הָעָם יַחְדָּו וַיֹּאמְרוּ, כֹּל אֲשֶׁר־ דִּבֶּר יְהוָה נַעֲשֶׂה"; [שמות ט"ו, ו'-ח']
כל יהודי זקוק לברכות. כדי לקבל אותם - חייב לקיים את התורה שכתובה בלשון הקודש:
"וְהָיָה, אִם שָׁמוֹעַ תִּשְׁמַע בְּקוֹל יְהוָה אֱלֹהֶיךָ, לִשְׁמֹר לַעֲשׂוֹת אֶת כָּל מִצְוÊתָיו, אֲשֶׁר אָנֹכִי מְצַוְּךָ הַיּוֹם--וּנְתָנְךָ יְהוָה אֱלֹהֶיךָ, עֶלְיוֹן, עַל, כָּל גּוֹיֵי הָאָרֶץ. וּבָאוּ עָלֶיךָ כָּל הַבְּרָכוֹת הָאֵלֶּה, וְהִשִּׂיגֻךָ: כִּי תִשְׁמַע, בְּקוֹל יְהוָה אֱלֹהֶיךָ. בָּרוּךְ אַתָּה, בָּעִיר; וּבָרוּךְ אַתָּה, בַּשָּׂדֶה".[להלן דברים כ"ח, א'- ד']
נבואת ישעיהו הנביא לעם ישראל - נבואת נחמה.
ישעיהו הנביא מנבא לעם ישראל כי עוד יבואו ימים טובים: "קוּמִי אוֹרִי, כִּי בָא אוֹרֵךְ; וּכְבוֹד יְהוָה, עָלַיִךְ זָרָח".[ישעיהו פרק: ס']
מהיכן בא האור ? מהתורה המשמשת - הגנה ומצילה את עם ישראל.
לסיכום, לאור האמור לעיל: כדי להתברך בברכות – ובשמחה ולנצח את האויבים נמלא את ציווי ה'- נהיה מאוחדים באהבת חינם - נזכה לכל הברכות כולל - עליונות על האויבים כנאמר:
"וַיהוָה הֶאֱמִירְךָ הַיּוֹם, לִהְיוֹת לוֹ לְעַם סְגֻלָּה, כַּאֲשֶׁר, דִּבֶּר ־ לָךְ; וְלִשְׁמֹר, כָּל מִצְוÊתָיו. וּלְתִתְּךָ עֶלְיוֹן, עַל כָּל־ הַגּוֹיִם אֲשֶׁר עָשָׂה, לִתְהִלָּה, וּלְשֵׁם וּלְתִפְאָרֶת; וְלִהְיֹתְךָ עַם קָדֹשׁ לַיהוָה אֱלֹהֶיךָ, כַּאֲשֶׁר דִּבֵּר"
כדברי ישעיהו הנביא: "קוּמִי אוֹרִי, כִּי בָא אוֹרֵךְ; וּכְבוֹד יְהוָה, עָלַיִךְ זָרָח"
[ישעיהו פרק: ס']
