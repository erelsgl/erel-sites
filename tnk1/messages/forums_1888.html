פרשת: ויצא- הבטחת ארץ ישראל לעם ישראל- לנצח?/ אהובה קליין

מאת: אהובה קליין

פרשת ויצא -הבטחת ארץ ישראל לעם ישראל – לנצח?
מאת: אהובה קליין .

פרשה זו פותחת בתיאור יציאת יעקב מבאר שבע - לחרן – כפי שכתוב: "וַיֵּצֵא יַעֲקֹב, מִבְּאֵר שָׁבַע; וַיֵּלֶךְ, חָרָנָה. וַיִּפְגַּע בַּמָּקוֹם וַיָּלֶן שָׁם, כִּי בָא הַשֶּׁמֶשׁ, וַיִּקַּח מֵאַבְנֵי הַמָּקוֹם, וַיָּשֶׂם מְרַאֲשֹׁתָיו; וַיִּשְׁכַּב, בַּמָּקוֹם הַהוּא. וַיַּחֲלֹם, וְהִנֵּה סֻלָּם מֻצָּב אַרְצָה, וְרֹאשׁוֹ, מַגִּיעַ הַשָּׁמָיְמָה; וְהִנֵּה מַלְאֲכֵי אֱלֹהִים, עֹלִים וְיֹרְדִים בּוֹ. וְהִנֵּה יְהוָה נִצָּב עָלָיו, וַיֹּאמַר, אֲנִי יְהוָה אֱלֹהֵי אַבְרָהָם אָבִיךָ, וֵאלֹהֵי יִצְחָק; הָאָרֶץ, אֲשֶׁר אַתָּה שֹׁכֵב עָלֶיהָ--לְךָ אֶתְּנֶנָּה, וּלְזַרְעֶךָ. וְהָיָה זַרְעֲךָ כַּעֲפַר הָאָרֶץ, וּפָרַצְתָּ יָמָּה וָקֵדְמָה וְצָפֹנָה וָנֶגְבָּה; וְנִבְרְכוּ בְךָ כָּל מִשְׁפְּחֹת הָאֲדָמָה, וּבְזַרְעֶךָ. וְהִנֵּה אָנֹכִי עִמָּךְ, וּשְׁמַרְתִּיךָ בְּכֹל אֲשֶׁר תֵּלֵךְ, וַהֲשִׁבֹתִיךָ, אֶל־ הָאֲדָמָה הַזֹּאת: כִּי, לֹא אֶעֱזָבְךָ, עַד אֲשֶׁר אִם עָשִׂיתִי, אֵת אֲשֶׁר דִּבַּרְתִּי לָךְ" [בראשית כ"ח', י'- ט"ז]
לקראת סוף הפרשה - שוב יעקב חולם חלום בעודו נמצא עם כל משפחתו בחרן כדברי הכתוב: "וַיֹּאמֶר אֵלַי מַלְאַךְ הָאֱלֹהִים, בַּחֲלוֹם--יַעֲקֹב; וָאֹמַר, הִנֵּנִי. וַיֹּאמֶר, שָׂא־ נָא עֵינֶיךָ וּרְאֵה כָּל־ הָעַתֻּדִים הָעֹלִים עַל הַצֹּאן, עֲקֻדִּים נְקֻדִּים, וּבְרֻדִּים: כִּי רָאִיתִי, אֵת כָּל־אֲשֶׁר לָבָן עֹשֶׂה לָּךְ. אָנֹכִי הָאֵל, בֵּית־ אֵל, אֲשֶׁר מָשַׁחְתָּ שָּׁם מַצֵּבָה, אֲשֶׁר נָדַרְתָּ לִּי שָׁם נֶדֶר; עַתָּה, קוּם צֵא מִן ־הָאָרֶץ הַזֹּאת, וְשׁוּב, אֶל־ אֶרֶץ מוֹלַדְתֶּךָ".[בראשית ל"א, י"א- י"ד]
השאלות הן:
א] במה ייחודו של יעקב - ומה משמעות החלום שחלם?
ב] מה ביקש יעקב מה' - ומה המסר לעם ישראל היום?
תשובות.
ייחודו של יעקב ומשמעות החלום שחלם.
רש"י שואל: הרי נאמר כבר בסוף הפרשה הקודמת-על יעקב [בפרשת תולדות] "וַיֵּלֶךְ פַּדֶּנָה אֲרָם" ? [בראשית, כ"ח, ז']
תשובתו : משום שהכתוב – הופסק שם - ובא לתאר: כי בנות כנען היו רעות בעיני יצחק אביו - הלך גם עשיו אל ישמעאל לקחת לו אישה משם ורק אחרי הפרטים על עשיו - כאן בפרשתנו הכתוב – שוב חוזר לספר על יציאת יעקב מבאר שבע לכיוון חרן,
עוד שואל רש"י מדוע נאמר: "וַיֵּצֵא יַעֲקֹב, מִבְּאֵר שָׁבַע? ולא נאמר עליו: "וילך"?
תשובת רש"י: כאשר צדיק יוצא ממקום הימצאותו למקום אחר - הדבר משפיע על המקום ממנו יוצא - ולמה? כי בזמן שהצדיק גר עדיין במקומו הקבוע, הרי הוא נחשב להוד והדר-כולו מאיר באור של צדיקים - לסביבה בו גר ,אך ברגע שהצדיק יוצא ממקומו באותו זמן ממש - הדבר משפיע על המקום שהיה גר בו - כי כל האור, הזיו ,ההוד והדר מתלווים אליו ממקום מושבו הקודם ואילו אותה עיר [באר שבע] נשארה מחוסרת ממעלתה וחשיבותה.
"ידוע כי ׃ יעקב אבינו נקרא בלשון חז"ל: "הבחיר שבאבות"- כדברי דוד המלך: "כִּֽי־ יַעֲקֹ֗ב בָּחַ֣ר ל֣וֹ יָ֑הּ יִ֝שְׂרָאֵ֗ל לִסְגֻלָּתֽוֹ" [תהלים: קל"ה, ד']
כאשר עם ישראל נמצא בצרה ומתפלל לבורא עולם - אלוקים מושיע אותנו בזכות יעקב הצדיק במילים: "יַֽעַנְךָ֣ יְ֭הוָה בְּי֣וֹם צָרָ֑ה יְ֝שַׂגֶּבְךָ֗ שֵׁ֤ם אֱלֹהֵ֬י יַֽעֲקֹֽב" [תהלים: כ', ב']
יעקב מלמד אותנו כיצד לצאת מן המצר, כנאמר:
"מִֽן־ הַ֭מֵּצַר קָרָ֣אתִי יָּ֑הּ עָנָ֖נִי בַמֶּרְחָ֣ב יָֽהּ" [תהלים : קי"ח, ה]
טרם תיאור חלום יעקב, נאמר: "וַיִּפְגַּע בַּמָּקוֹם וַיָּלֶן שָׁם, כִּי־ בָא הַשֶּׁמֶשׁ"
רש"י מעיר: הרי לא נאמר באיזה מקום מדובר. אלא נאמר: "בַּמָּקוֹם" - כמו בהמקום- בה"א הידיעה - המשמעות היא: כי הדבר נזכר כבר במקום אחר כפי שכתוב:"...... וַיִּשָּׂא אַבְרָהָם אֶת־ עֵינָיו וַיַּרְא אֶת ־הַמָּקוֹם—מֵרָחֹק" [בראשית: כ"ב ,ד'] כאשר אברהם בדרך לעקדת יצחק ראה מרחוק את הר המוריה, וידוע שהר המוריה- הוא מקום בית המקדש.
חז"ל אומרים על המילה "וַיִּפְגַּע"- ג' פירושים:
א] לשון פגישה - שיעקב הגיע וניפגש עם המקום,
ב] "וַיִּפְגַּע"- פירושו: התפלל - יעקב התפלל במקום הקדוש - ותיקן שם את "תפילת ערבית".
ג] "וַיִּפְגַּע"— התקצרה לו הדרך- נעקר הר המוריה ממקומו - והדרך לשם באה לקראתו ועוד נאמר: כיון שהגיע לשם יעקב שקעה לו החמה טרם זמנה - כדי שילון דווקא במקום זה.
חלומו של יעקב- "וַיַּחֲלֹם, וְהִנֵּה סֻלָּם מֻצָּב אַרְצָה, וְרֹאשׁוֹ, מַגִּיעַ הַשָּׁמָיְמָה; וְהִנֵּה מַלְאֲכֵי אֱלֹהִים, עֹלִים וְיֹרְדִים בּוֹ".
חלום זה של יעקב- מצטייר כעין מעמד הר סיני - לפי- שבזמן קבלת התורה כולם התייצבו לרגלי ההר כפי שנאמר: "וַיִּתְיַצְּבוּ, בְּתַחְתִּית הָהָר" [שמות י"ט, י"ז] ובתיאור חלום יעקב נאמר: "..סֻלָּם מֻצָּב אַרְצָה" ובמעמד הר סיני נאמר: "וַתִּקְרְבוּן וַתַּעַמְדוּן, תַּחַת הָהָר; וְהָהָר בֹּעֵר בָּאֵשׁ, עַד ־ לֵב הַשָּׁמַיִם" [דברים ,ד', י"א] ואילו בחלום יעקב נאמר: "וְרֹאשׁוֹ, מַגִּיעַ הַשָּׁמָיְמָה";
במעמד הר סיני נאמר: "אָנֹכִי יְהוָה אֱלֹהֶיךָ,," [שמות" כ', ב' ,]
ובחלום יעקב- "אֲנִי יְהוָה אֱלֹהֵי אַבְרָהָם אָבִיךָ"
[בהשראת הספר: "ויאמר ראובן"/ ראובן בן דוד יוספוב]
בחלום הסולם - ה' מבטיח ליעקב ובניו את ההבטחה הנצחית - ארץ ישראל והבטחת הבנים:
"הָאָרֶץ, אֲשֶׁר אַתָּה שֹׁכֵב עָלֶיהָ--לְךָ אֶתְּנֶנָּה, וּלְזַרְעֶךָ. וְהָיָה זַרְעֲךָ כַּעֲפַר הָאָרֶץ, וּפָרַצְתָּ יָמָּה וָקֵדְמָה וְצָפֹנָה וָנֶגְבָּה; וְנִבְרְכוּ בְךָ כָּל מִשְׁפְּחֹת הָאֲדָמָה, וּבְזַרְעֶךָ".
רש" מסביר: הָאָרֶץ, אֲשֶׁר אַתָּה שֹׁכֵב עָלֶיהָ"- קיפל הקב"ה את כל ארץ ישראל לתוך ארבע האמות שהיו תחתיו וכך הבטיח לו את כל ארץ ישראל שתהיה נוחה להיכבש על ידי בניו.
רש"י מבאר: כי יעקב הניח את האבנים סביב ראשו בצורת חצי עיגול- כעין מרזב- צינור מעוגל שדרכו זורמים מי הגשמים מהגג - כהגנה על ראשו מפני חיות רעות שלא יפגעו בו בשנתו.
התחילו האבנים מריבות זו עם זו - לפי שכל אחת ביקשה - שהצדיק יניח עליה את ראשו- מיד ה' חיבר אותן לאבן אחת ומכאן שעל כולן -יחד - יעקב הניח את ראשו וההוכחה שנאמר:
"וַיַּשְׁכֵּם יַעֲקֹב בַּבֹּקֶר, וַיִּקַּח אֶת־ הָאֶבֶן אֲשֶׁר ־שָׂם מְרַאֲשֹׁתָיו" [בראשית כ"ח, י"ח]
לא נאמר: אבנים- אלא": אבן!
יש כאן רמז לאחדות לפי שנאמר:"וַיִּקַּח מֵאַבְנֵי הַמָּקוֹם, וַיָּשֶׂם מְרַאֲשֹׁתָיו"
על פי החלום –לעניות דעתי, המסרים הם:
א] ארץ ישראל מובטחת לאבותינו - ולכל הצאצאים אחריהם - כלומר לכל עם ישראל- לדורי דורות ,אך מעמד הר סיני מזכיר לנו: כי אנחנו יהודים - ובארץ ישראל הובטחה לנו בתנאי: שנקיים את הבטחתנו- כאיש אחד בלב אחד לרגלי הר סיני- כנאמר: "וַיִּקַּח סֵפֶר הַבְּרִית וַיִּקְרָא בְּאׇזְנֵי הָעָם וַיֹּאמְרוּ כֹּל אֲשֶׁר דִּבֶּר יְהֹוָה נַעֲשֶׂה וְנִשְׁמָע" [שמות כ"ד, זי]
ב] רש"י סובר: כי המלאכים המופיעים בחלום - עולים ויורדים על גבי הסולם - אותם מלאכים שעולים תחילה בשלבי הסולם כלפי שמים - הם המלאכים שליוו את יעקב בארץ ואין הם יוצאים לחו"ל ובמקומם יורדים המלאכים שילוו את יעקב להמשך דרכו-מעבר לגבול הארץ.
בנוסף למלאכים השומרים על יעקב בכל מקום הרי אלוקים שומר עליו מלמעלה!
כך גם ה' שומר על עם ישראל תמיד – כדברי דוד המלך:
"כִּ֤י לֹֽא ־ יִטֹּ֣שׁ יְהוָ֣ה עַמּ֑וֹ וְ֝נַֽחֲלָת֗וֹ לֹ֣א יַֽעֲזֹֽב" [תהלים צ"ד, י"ד]
ג] על פי התיאור של האבנים שיעקב הניח סביב ראשו - והם התאחדו לאבן אחת – כדי שכל האבנים ייקחו חלק בשמירה על יעקב הצדיק, נשאל: אם אבנים שהן נחשבות לדומם - מסוגלות להתאחד - קל וחומר שעם ישראל חייב להתאחד - כאיש אחד בלב אחד- למען נזכה לעמוד באופן איתן כנגד כל האויבים בהצלחה! בסייעתא דשמיא.
יעקב מבקש – מאלוקים בקשה צנועה.
"וַיִּדַּר יַעֲקֹב, נֶדֶר לֵאמֹר: אִם יִהְיֶה אֱלֹהִים עִמָּדִי, וּשְׁמָרַנִי בַּדֶּרֶךְ הַזֶּה אֲשֶׁר אָנֹכִי הוֹלֵךְ, וְנָתַן לִי לֶחֶם לֶאֱכֹל, וּבֶגֶד לִלְבֹּשׁ. וְשַׁבְתִּי בְשָׁלוֹם, אֶל בֵּית אָבִי; וְהָיָה יְהוָה לִי, לֵאלֹהִים. וְהָאֶבֶן הַזֹּאת, אֲשֶׁר שַׂמְתִּי מַצֵּבָה--יִהְיֶה, בֵּית אֱלֹהִים; וְכֹל אֲשֶׁר תִּתֶּן לִי, עַשֵּׂר אֲעַשְּׂרֶנּוּ לָךְ".[ להלן, כ"ח, כ'-כ"ב]
יעקב לא ביקש כסף ולא זהב, אלא התכוון לבקש בריאות, שהרי רק אדם בריא יכול ללבוש בגדים נאים, ולאכול מזון בכוחות עצמו. הוא ביקש לשוב למקום בו גדל – בריא ושלם. וביקש כי המקום הזה – יהיה בית אלוקים ויקריב לה' קורבנות וייתן מעשרות.
הרש"ר- הרב שמשון רפאל הירש מסביר: האבן הזאת- אבן הפינה של בית המקדש הראשון אשר תונח שנית בבית המקדש השני. אך הגמר הסופי של הבניין -"אבן השתייה"- תהיה יסוד בית המקדש השלישי.
לסיכום לאור האמור לעיל: ארץ ישראל הובטחה רק לעם ישראל לעד!
הן בחלום בתחילת הפרשה והן בחלום לקראת סוף הפרשה.
אך כדי שנהיה ראויים לחיות בארצנו הקדושה לעד בשלווה ובשלום, עלינו לקיים את השליחות שלנו בארץ - לנצח - לקיים את עשרת הדברות - כולל שמירת שבת כמו שכתוב:
"וְעַתָּה יִשְׂרָאֵל, שְׁמַע אֶל־ הַחֻקִּים וְאֶל־ הַמִּשְׁפָּטִים, אֲשֶׁר אָנֹכִי מְלַמֵּד אֶתְכֶם, לַעֲשׂוֹת--לְמַעַן תִּחְיוּ, וּבָאתֶם וִירִשְׁתֶּם אֶת־ הָאָרֶץ, אֲשֶׁר יְהוָה אֱלֹהֵי אֲבֹתֵיכֶם, נֹתֵן לָכֶם. לֹא תֹסִפוּ, עַל הַדָּבָר אֲשֶׁר אָנֹכִי מְצַוֶּה אֶתְכֶם, וְלֹא תִגְרְעוּ, מִמֶּנּוּ--לִשְׁמֹר, אֶת מִצְוÊת יְהוָה אֱלֹהֵיכֶם, אֲשֶׁר אָנֹכִי, מְצַוֶּה אֶתְכֶם" [דברים .ד', א'- ג']
כל מה שהבטיח ה' ליעקב - זה אמור גם על כל עם ישראל.
בימים אלה ממש כאשר - המחלוקת עדיין אוחזת בתוך העם – כאש בשדה קוצים, על ידי קבוצה מצומצמת של אנשים בתוך העם המעוניין במדינה ללא תורה, חלילה ולהתחבר לאויבים האכזרים ורוצחים - עלינו להתחזק באמונה ובקיום המצוות כולל שמירת שבת - שהיא אחת מעשרת הדברות - ולשאת תפילה לה' כדוגמת יעקב שהתפלל אל ה' וביקש לשוב לשלום – בריא ושלם לארץ המובטחת.
כדוגמת דוד המלך:
"ק֘וּמָ֤ה יְהוָ֨ה ׀ הֽוֹשִׁ֘יעֵ֤נִי אֱלֹהַ֗י כִּֽי הִכִּ֣יתָ אֶת־ כָּל ־ אֹֽיְבַ֣י לֶ֑חִי שִׁנֵּ֖י רְשָׁעִ֣ים שִׁבַּֽרְתָּ׃ ליהוָ֥ה הַיְשׁוּעָ֑ה עַֽל ־ עַמְּךָ֖ בִרְכָתֶ֣ךָ סֶּֽלָה" [תהלים ג, ח'- טי] אמן ואמן.
